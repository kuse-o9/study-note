# Gemini提案の実コード照合・追加修正

2026-11-08。Gemini側はGitHubを読めなかったため、提出された文章は一般的な仮説として扱った。実コードと公式資料を照合し、機能・画質・乗客数・保存形式を維持して修正した。課金サービス・AI APIの追加はない。

## 採用と訂正

1. 鉄道はCesium 1.117、こもれび島はThree.jsを使用。CesiumのPrimitiveCollection.removeは既定でdestroyする。Three.jsのmaterial.disposeを鉄道に一律適用しない。
2. こもれび島のチャンクは共有パレットを使うため、従来のgeometry解放を維持。利用者が建物を解体した場合には、専用geometry・material・textureを解放する。場面内または共有パレットで使用中の資源は解放しない。renderer所有の影バッファを各建物の解放対象にしない。
3. あずさ・かいじの生成模型を協調的なgeneratorで構築。約6msの処理区切りを目安にsetTimeoutで制御を戻す。非表示タブでもrequestAnimationFrame待ちで永遠に止まらない。従来の同期build APIも残す。
4. 古い場面の構築は世代番号で中止し、破棄した場面へ模型を追加しない。GPU停止時には生成処理と一時キャッシュを破棄する。
5. 共通のcontextlost処理は既にpreventDefaultしていた。今回は描画ループの即時停止と重複保存防止、復旧通知を追加。復旧イベントだけで壊れたViewerを自動再開しない。保存後、利用者が再読み込みで作り直す。特急の既存の新規Viewer再構築経路も維持。
6. preventDefaultは復旧可能性を残す処理であり、GPUブロック回避や復旧成功を保証しない。Disabledという文字だけでGPUドライバーのクラッシュやVRAM枯渇を断定しない。CPUの50ms超処理も、それだけでOSのGPUタイムアウト発動を示さない。

## 診断機能

3Dエラー画面に「診断情報をコピー」を追加。最新40件の開始失敗・喪失・復旧イベント、長いタスクの回数・時間、描画済みフレーム数と模型オブジェクト数、生成模型の構築統計を取得する。

集計はそのページのメモリー内だけで保持し、サーバーへ自動送信・永続保存しない。利用者がボタンを押したときだけJSONをコピー。クリップボードが許可されない環境では読み取り専用のコピー欄を表示する。URL、トークン、メール、アカウント情報、保存データ、エラー本文をJSONへ入れない。オブジェクト数はVRAMのバイト数ではない。要求時描画なので、毎秒描画回数の低さだけで処理落ちと断定しない。

## 検査と限界

- 変更元18ファイルをGitHubの公開元blobと照合。
- model-timeslicing.test.cjs：あずさ・かいじ各4模型（線路・一般駅・終端配線）のハッシュが修正前の固定値と一致。頂点、色、索引、乗客データを含む。各20回の小模型の構築・破棄、構築中止、同じ模型の重複要求共有を確認。
- 模擬CPU検査では、あずさ166回、かいじ150回ブラウザー相当のホスト処理へ制御を返した。一方、最長区切りは約377ms／194msで、全Long Taskをなくしたわけではない。単一の重い操作・GC等の追加調査が必要。実機FPSやTDR改善量ではない。
- island-owned-release.test.cjs：20回の模擬解放で、専用資源を解放し、表示中・共有パレットの資源を維持。
- island-chunk-release.test.cjs：既存チャンク解放と7×7表示範囲を維持。
- recovery-lifecycle.test.cjs：20回の重複喪失／描画エラーで一度だけ保存、描画停止、手動復旧、破棄済みViewer無視。
- scene-diagnostics.test.cjs：診断件数上限、コピー、機密値・URL等を含めないこと。
- gpu-blocked-start.test.cjs・scene-recovery.test.cjs：ブロック時の無駄な再試行防止、正しいViewerの一度だけ解放、保存順序を確認。

- 確認ブラウザーはGPU無効のため、実機のGPU復旧、VRAM推移、長時間走行での再発防止は未確認。診断を根拠に次を絞り込む。

/* Bound temporary model data and loading bursts without changing visible geometry. */
(function(g){'use strict';
 function create({maxBytes=32*1024*1024,concurrency=2,timeoutMs=25000}={}){
  if(!(maxBytes>0)||!Number.isInteger(concurrency)||concurrency<1||!(timeoutMs>0))throw Error('Invalid asset limits');
  const weights=new WeakMap(),entries=new Map(),queue=[];let bytes=0,active=0,peakActive=0,evictions=0;
  function sizeOf(value){
   if(value&&typeof value==='object'&&weights.has(value))return weights.get(value);
   const seen=new WeakSet(),stack=[value];let size=0;
   while(stack.length){const item=stack.pop();if(typeof item==='string'){size+=item.length*2;continue;}if(typeof item==='number'){size+=8;continue;}if(!item||typeof item!=='object')continue;if(seen.has(item))continue;seen.add(item);size+=32;
    if(ArrayBuffer.isView(item)){size+=item.byteLength;continue;}if(item instanceof ArrayBuffer){size+=item.byteLength;continue;}
    for(const key of Object.keys(item)){size+=key.length*2+8;stack.push(item[key]);}
   }
   if(value&&typeof value==='object')weights.set(value,size);return size;
  }
  function remove(key){const old=entries.get(key);if(!old)return false;bytes-=old.bytes;return entries.delete(key);}
  function pump(){while(active<concurrency&&queue.length){const job=queue.shift();active++;peakActive=Math.max(peakActive,active);Promise.resolve().then(job.work).then(job.resolve,job.reject).finally(()=>{active--;pump();});}}
  const api={get(key){return entries.get(key)?.value;},has:key=>entries.has(key),keys:()=>entries.keys(),delete:remove,clear(){entries.clear();bytes=0;},get size(){return entries.size;},
   set(key,value){remove(key);const size=sizeOf(value);if(size<=maxBytes){entries.set(key,{value,bytes:size});bytes+=size;}while(bytes>maxBytes){remove(entries.keys().next().value);evictions++;}return api;},
   run(work){return new Promise((resolve,reject)=>{queue.push({work,resolve,reject});pump();});},
   async fetchJSON(url){const controller=new AbortController(),timer=setTimeout(()=>controller.abort(),timeoutMs);try{const response=await g.fetch(url,{signal:controller.signal});if(!response.ok)throw Error('model fetch: '+response.status);return await response.json();}catch(error){if(controller.signal.aborted)throw Error('模型データの通信が時間内に完了しませんでした。再試行できます。');throw error;}finally{clearTimeout(timer);}},
   getStats:()=>({bytes,maxBytes,entries:entries.size,active,queued:queue.length,peakActive,evictions})
  };return api;
 }

})(globalThis);

function cartAt(d,lift=0,g=geo){const p=C.position(g,d);return Ces.Cartesian3.fromDegrees(p.lon,p.lat,p.height+lift);}
function surfaceFrame(g,d,span=.3){const rear=cartAt(d-span,0,g),front=cartAt(d+span,0,g),forward=Ces.Cartesian3.normalize(Ces.Cartesian3.subtract(front,rear,new Ces.Cartesian3()),new Ces.Cartesian3()),center=cartAt(d,0,g),radial=Ces.Ellipsoid.WGS84.geodeticSurfaceNormal(center,new Ces.Cartesian3()),left=Ces.Cartesian3.normalize(Ces.Cartesian3.cross(radial,forward,new Ces.Cartesian3()),new Ces.Cartesian3()),up=Ces.Cartesian3.normalize(Ces.Cartesian3.cross(forward,left,new Ces.Cartesian3()),new Ces.Cartesian3());return {center,forward,left,up};}
function rotationFor(frame){const rotation=new Ces.Matrix3();Ces.Matrix3.setColumn(rotation,0,frame.forward,rotation);Ces.Matrix3.setColumn(rotation,1,frame.left,rotation);Ces.Matrix3.setColumn(rotation,2,frame.up,rotation);return Ces.Quaternion.fromRotationMatrix(rotation,new Ces.Quaternion());}
function carPose(d,g=geo){const frame=surfaceFrame(g,d,7.1);const front=cartAt(d+7.1,0,g),rear=cartAt(d-7.1,0,g);frame.center=Ces.Cartesian3.midpoint(front,rear,new Ces.Cartesian3());let rise=0;for(const offset of [-9.75,-7.1,0,7.1,9.75]){const road=cartAt(d+offset,0,g),line=Ces.Cartesian3.add(frame.center,Ces.Cartesian3.multiplyByScalar(frame.forward,offset,new Ces.Cartesian3()),new Ces.Cartesian3());rise=Math.max(rise,Ces.Cartesian3.dot(Ces.Cartesian3.subtract(road,line,new Ces.Cartesian3()),frame.up));}frame.center=Ces.Cartesian3.add(frame.center,Ces.Cartesian3.multiplyByScalar(frame.up,rise+.15,new Ces.Cartesian3()),new Ces.Cartesian3());return frame;}
function lifted(frame,height){return Ces.Cartesian3.add(frame.center,Ces.Cartesian3.multiplyByScalar(frame.up,height,new Ces.Cartesian3()),new Ces.Cartesian3());}
